cmake_minimum_required(VERSION 3.16)
project(f1_command_monitor)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
add_executable(f1_command_monitor monitor.cpp)
ament_target_dependencies(f1_command_monitor rclcpp sensor_msgs control_msgs geometry_msgs)
add_library(f1_command_monitor_shared SHARED monitor_shared.cpp)
ament_target_dependencies(f1_command_monitor_shared rclcpp sensor_msgs control_msgs geometry_msgs)
